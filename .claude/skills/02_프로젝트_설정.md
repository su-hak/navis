## ⚙️ 4단계: 핵심 파일 설정

### main.ts 설정
```typescript
import { NestFactory } from '@nestjs/core';
import { ValidationPipe } from '@nestjs/common';
import { AppModule } from './app.module';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);

  // CORS 설정
  app.enableCors({
    origin: process.env.NODE_ENV === 'production'
      ? process.env.FRONTEND_URL || 'https://yourdomain.com'
      : '*',
    credentials: true,
  });

  // Validation Pipe (글로벌)
  app.useGlobalPipes(new ValidationPipe({
    whitelist: true,        // DTO에 없는 속성 제거
    forbidNonWhitelisted: true, // DTO에 없는 속성 전송 시 에러
    transform: true,        // 타입 자동 변환
  }));

  const port = process.env.PORT || 3000;
  await app.listen(port);
  console.log(`🚀 Application running on: http://localhost:${port}`);
}
bootstrap();
```

### app.module.ts 설정 (TypeORM)
```typescript
import { Module } from '@nestjs/common';
import { ConfigModule, ConfigService } from '@nestjs/config';
import { TypeOrmModule } from '@nestjs/typeorm';
import { ScheduleModule } from '@nestjs/schedule';

@Module({
  imports: [
    // 환경변수 글로벌 설정
    ConfigModule.forRoot({
      isGlobal: true,
      envFilePath: '.env',
    }),

    // TypeORM 설정 (환경변수 우선순위 패턴)
    TypeOrmModule.forRootAsync({
      imports: [ConfigModule],
      useFactory: (configService: ConfigService) => {
        // Railway 환경변수 우선, 없으면 커스텀 환경변수, 없으면 기본값
        const host =
          configService.get<string>('MYSQLHOST') ||
          configService.get<string>('DB_HOST') ||
          'localhost';
        const port = parseInt(
          configService.get<string>('MYSQLPORT') ||
            configService.get<string>('DB_PORT') ||
            '3306',
          10,
        );
        const username =
          configService.get<string>('MYSQLUSER') ||
          configService.get<string>('DB_USERNAME') ||
          'root';
        const password =
          configService.get<string>('MYSQLPASSWORD') ||
          configService.get<string>('DB_PASSWORD') ||
          '';
        const database =
          configService.get<string>('MYSQLDATABASE') ||
          configService.get<string>('DB_DATABASE') ||
          'myapp';

        return {
          type: 'mysql',
          host,
          port,
          username,
          password,
          database,
          entities: [__dirname + '/**/*.entity{.ts,.js}'],
          synchronize: process.env.NODE_ENV !== 'production', // 프로덕션: false
          logging: process.env.NODE_ENV === 'development',
        };
      },
      inject: [ConfigService],
    }),

    // 스케줄러 (정기 결제 등)
    ScheduleModule.forRoot(),

    // 모듈 추가 (순서대로)
    // AuthModule,
    // UsersModule,
    // SubscriptionModule,
    // AbTestModule,
  ],
})
export class AppModule {}
```

## 🔑 5단계: 환경변수 우선순위 패턴

### 핵심 패턴
```typescript
// Railway, Heroku, Vercel 등 배포 환경마다 환경변수 이름이 다름
// → 우선순위 기반 Fallback 패턴 사용

const host =
  configService.get<string>('MYSQLHOST') ||      // Railway
  configService.get<string>('DB_HOST') ||        // 커스텀
  'localhost';                                    // 기본값
```

### 모든 프로젝트에 적용
- Railway: `MYSQLHOST`, `MYSQLPORT`, `MYSQLUSER`, `MYSQLPASSWORD`, `MYSQLDATABASE`
- Heroku: `DATABASE_URL` (파싱 필요)
- 로컬: `DB_HOST`, `DB_PORT`, `DB_USERNAME`, `DB_PASSWORD`, `DB_DATABASE`
