## 💳 Payment Entity (완전판)

```typescript
import {
  Entity,
  PrimaryGeneratedColumn,
  Column,
  ManyToOne,
  CreateDateColumn,
  JoinColumn,
  Index,
} from 'typeorm';
import { User } from './user.entity';

export type PaymentStatus = 'success' | 'failed' | 'cancelled';

@Entity('payments')
@Index('idx_user_created', ['user', 'createdAt']) // 조회 최적화
export class Payment {
  @PrimaryGeneratedColumn('uuid')
  id: string;

  @ManyToOne(() => User, { onDelete: 'CASCADE' })
  @JoinColumn({ name: 'user_id' })
  user: User;

  @Column({ type: 'varchar', length: 100 })
  tid: string; // 거래ID (Transaction ID)

  @Column({ type: 'varchar', length: 100 })
  orderId: string; // 주문번호

  @Column({ type: 'decimal', precision: 10, scale: 2 })
  amount: number; // 결제금액

  @Column({ type: 'enum', enum: ['success', 'failed', 'cancelled'] })
  status: PaymentStatus;

  @Column({ type: 'varchar', length: 500, nullable: true })
  failReason: string | null; // 실패 원인

  @CreateDateColumn({ name: 'created_at' })
  createdAt: Date;
}
```

**핵심 포인트**:
- ✅ 성공/실패 모두 저장 (실패 원인 포함)
- ✅ `decimal` 타입으로 금액 저장 (정확도)
- ✅ 복합 인덱스로 조회 최적화

### Basic 인증 헤더
```typescript
private get basicAuth(): string {
  const clientId = this.configService.get<string>('NICEPAY_CLIENT_ID') || '';
  const secretKey = this.configService.get<string>('NICEPAY_SECRET_KEY') || '';
  return `Basic ${Buffer.from(`${clientId}:${secretKey}`).toString('base64')}`;
}
```

### 빌링키 등록
```typescript
async registerBillingKey(authToken: string, orderId: string): Promise<string> {
  const response = await fetch(`${this.apiUrl}/subscribe/regist`, {
    method: 'POST',
    headers: {
      Authorization: this.basicAuth,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      encData: authToken,
      orderId,
      returnCharSet: 'utf-8',
    }),
  });

  const data = await response.json();

  if (data.resultCode !== '0000') {
    throw new BadRequestException(`빌링키 등록 실패: ${data.resultMsg}`);
  }

  return data.bid; // 빌링키 반환
}
```

### 빌링키로 결제
```typescript
async chargeWithBillingKey(
  bid: string,
  orderId: string,
  amount: number,
): Promise<string> {
  const response = await fetch(`${this.apiUrl}/subscribe/payments`, {
    method: 'POST',
    headers: {
      Authorization: this.basicAuth,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      bid,
      orderId,
      amount,
      goodsName: 'StockSheet Pro',
      cardQuota: '00', // 일시불
      returnCharSet: 'utf-8',
    }),
  });

  const data = await response.json();

  if (data.resultCode !== '0000') {
    throw new BadRequestException(`결제 실패: ${data.resultMsg}`);
  }

  return data.tid; // 거래ID 반환
}
```

### 빌링키 삭제
```typescript
async deleteBillingKey(bid: string): Promise<void> {
  try {
    await fetch(`${this.apiUrl}/subscribe/${bid}`, {
      method: 'DELETE',
      headers: { Authorization: this.basicAuth },
    });
  } catch (error) {
    this.logger.warn(`빌링키 삭제 실패 (무시): ${error.message}`);
  }
}
```

## 🔐 카드 정보 암호화 패턴

### AES-128 암호화
```typescript
import { createCipheriv } from 'crypto';

private encryptCardData(plaintext: string): string {
  const secretKey = this.configService.get<string>('NICEPAY_SECRET_KEY') || '';
  if (secretKey.length < 16) {
    throw new InternalServerErrorException('NICEPAY_SECRET_KEY가 너무 짧습니다.');
  }

  const key = secretKey.substring(0, 16);
  const cipher = createCipheriv('aes-128-ecb', Buffer.from(key, 'utf-8'), null);

  let encrypted = cipher.update(plaintext, 'utf-8', 'hex');
  encrypted += cipher.final('hex');

  return encrypted;
}
```

## 📊 결제 내역 조회 API

### PaymentController (선택)
```typescript
@Controller('api/payments')
@UseGuards(JwtAuthGuard)
export class PaymentController {
  constructor(private readonly paymentService: PaymentService) {}

  // 내 결제 내역 조회
  @Get('my')
  async getMyPayments(@Request() req) {
    const userId = req.user.id;
    return this.paymentService.findByUserId(userId);
  }

  // 결제 내역 상세 조회
  @Get(':id')
  async getPaymentDetail(@Request() req, @Param('id') id: string) {
    const userId = req.user.id;
    return this.paymentService.findById(id, userId);
  }
}
```

### PaymentService
```typescript
async findByUserId(userId: string): Promise<Payment[]> {
  return this.paymentRepository.find({
    where: { user: { id: userId } },
    order: { createdAt: 'DESC' },
  });
}

async findById(id: string, userId: string): Promise<Payment> {
  const payment = await this.paymentRepository.findOne({
    where: { id, user: { id: userId } },
  });

  if (!payment) {
    throw new NotFoundException('결제 내역을 찾을 수 없습니다.');
  }

  return payment;
}
```
