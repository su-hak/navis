// ✅ 좋은 패턴: 빌링키만 삭제, 만료일까지 Pro 유지

async prepare(userId: string): Promise<{ orderId: string }> {
  const timestamp = Date.now();
  const orderId = `ORDER_${userId}_${timestamp}`;
  return { orderId };
}

### 2. 빌링키 등록 + 첫 결제
```typescript
async approve(userId: string, dto: ApproveSubscriptionDto) {
  const user = await this.userRepository.findOne({ where: { id: userId } });

  // 1. 빌링키 등록 (결제 게이트웨이 API 호출)
  const bid = await this.registerBillingKey(dto.authToken, dto.orderId);

  // 2. 빌링키로 첫 결제 실행
  const tid = await this.chargeWithBillingKey(bid, dto.orderId, 9900);

  // 3. DB 업데이트
  const expiresAt = new Date();
  expiresAt.setMonth(expiresAt.getMonth() + 1);

  user.plan = 'pro';
  user.billingKey = bid;
  user.subscriptionExpiresAt = expiresAt;
  await this.userRepository.save(user);

  // 4. 결제 이력 저장
  await this.paymentRepository.save({
    user,
    tid,
    orderId: dto.orderId,
    amount: 9900,
    status: 'success',
  });

  return { success: true, message: 'Pro 구독이 시작되었습니다.' };
}
```

### 3. 구독 취소 (만료일까지 유지)
```typescript
async cancel(userId: string) {
  const user = await this.userRepository.findOne({ where: { id: userId } });

  if (user.plan !== 'pro') {
    throw new BadRequestException('Pro 구독 중이 아닙니다.');
  }

  // 빌링키 삭제 (결제 게이트웨이 API 호출)
  if (user.billingKey) {
    await this.deleteBillingKey(user.billingKey);
  }

  // ✅ 핵심: billingKey만 삭제, plan은 유지
  user.billingKey = null;
  await this.userRepository.save(user);

  const expiresAt = user.subscriptionExpiresAt;
  const expiresStr = expiresAt
    ? `${expiresAt.getFullYear()}년 ${expiresAt.getMonth() + 1}월 ${expiresAt.getDate()}일`
    : '';

  return {
    success: true,
    message: `구독이 취소되었습니다. ${expiresStr}까지 Pro 기능을 사용할 수 있습니다.`,
  };
}
```

### 5. 정기 결제 스케줄러
```typescript
import { Cron } from '@nestjs/schedule';
import { LessThanOrEqual } from 'typeorm';

@Cron('0 9 * * *') // 매일 오전 9시
async processRecurringBilling() {
  const today = new Date();
  const todayStart = new Date(today.getFullYear(), today.getMonth(), today.getDate());
  const todayEnd = new Date(today.getFullYear(), today.getMonth(), today.getDate() + 1);

  // 오늘 만료 예정인 Pro 유저 중 billingKey가 있는 유저
  const usersToCharge = await this.userRepository.find({
    where: {
      plan: 'pro',
      subscriptionExpiresAt: LessThanOrEqual(todayEnd),
    },
  });

  const activeUsers = usersToCharge.filter(
    (u) => u.billingKey && u.subscriptionExpiresAt >= todayStart,
  );

  for (const user of activeUsers) {
    await this.chargeRecurring(user);
  }

  // 만료된 유저 plan 'free'로 변경
  await this.expireSubscriptions();
}

private async chargeRecurring(user: User) {
  const orderId = `RECURRING_${user.id.substring(0, 8)}_${Date.now()}`;

  try {
    // 빌링키로 결제 실행
    const tid = await this.chargeWithBillingKey(user.billingKey, orderId, 9900);

    // 결제 성공: 만료일 1개월 연장
    const newExpiry = new Date(user.subscriptionExpiresAt);
    newExpiry.setMonth(newExpiry.getMonth() + 1);
    user.subscriptionExpiresAt = newExpiry;
    await this.userRepository.save(user);

    // 결제 이력 저장
    await this.paymentRepository.save({
      user,
      tid,
      orderId,
      amount: 9900,
      status: 'success',
    });

    this.logger.log(`정기결제 성공: ${user.email}`);
  } catch (error) {
    // 결제 실패 시 plan을 'free'로 변경
    user.plan = 'free';
    user.billingKey = null;
    user.subscriptionExpiresAt = null;
    await this.userRepository.save(user);

    this.logger.warn(`정기결제 실패: ${user.email}`);
  }
}

private async expireSubscriptions() {
  const now = new Date();

  // billingKey가 없고 만료일이 지난 Pro 유저 → free로 변경
  const expiredUsers = await this.userRepository.find({
    where: { plan: 'pro' },
  });

  const toExpire = expiredUsers.filter(
    (u) => !u.billingKey && u.subscriptionExpiresAt && u.subscriptionExpiresAt < now,
  );

  for (const user of toExpire) {
    user.plan = 'free';
    user.subscriptionExpiresAt = null;
    await this.userRepository.save(user);
    this.logger.log(`구독 만료 처리: ${user.email}`);
  }
}
```

## 🎮 Subscription Controller

### subscription.controller.ts
```typescript
import {
  Controller,
  Post,
  Get,
  Body,
  UseGuards,
  Request,
} from '@nestjs/common';
import { SubscriptionService } from './subscription.service';
import { ApproveSubscriptionDto } from './dto/approve-subscription.dto';
import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';

@Controller('api/subscription')
export class SubscriptionController {
  constructor(private readonly subscriptionService: SubscriptionService) {}

  // orderId 생성
  @Post('prepare')
  @UseGuards(JwtAuthGuard)
  async prepare(@Request() req: any) {
    return this.subscriptionService.prepare(req.user.id);
  }

  // 빌링키 등록 + 첫 결제 승인
  @Post('approve')
  @UseGuards(JwtAuthGuard)
  async approve(@Request() req: any, @Body() dto: ApproveSubscriptionDto) {
    return this.subscriptionService.approve(req.user.id, dto);
  }

  // 구독 취소
  @Post('cancel')
  @UseGuards(JwtAuthGuard)
  async cancel(@Request() req: any) {
    return this.subscriptionService.cancel(req.user.id);
  }

  // 구독 상태 조회
  @Get('status')
  @UseGuards(JwtAuthGuard)
  async getStatus(@Request() req: any) {
    return this.subscriptionService.getStatus(req.user.id);
  }
}
```
